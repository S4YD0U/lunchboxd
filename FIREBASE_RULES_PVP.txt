// ============================================================
// RÈGLES FIRESTORE — À ajouter dans ta console Firebase
// Console Firebase → Firestore → Règles
// ============================================================
//
// Ajoute ces règles à tes règles existantes.
// La collection "duels" permet :
//   - Créer un duel si connecté
//   - Lire un duel si on en est participant (host ou guest)
//   - Mettre à jour un duel si on en est participant
//   - Pas de suppression (pour l'historique)
//
// ============================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─── Règles existantes (users) ──────────────────────────────
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // ─── Règles pour les duels PvP ──────────────────────────────
    match /duels/{duelId} {
      // Créer un duel : doit être connecté, et être l'hôte
      allow create: if request.auth != null
        && request.resource.data.host.uid == request.auth.uid;

      // Lire un duel : doit être connecté
      // (on ne vérifie pas si participant pour le join initial)
      allow read: if request.auth != null;

      // Mettre à jour : doit être host ou guest du duel
      allow update: if request.auth != null && (
        resource.data.host.uid == request.auth.uid ||
        (resource.data.guest != null && resource.data.guest.uid == request.auth.uid)
      );

      // Pas de suppression
      allow delete: if false;
    }

  }
}
